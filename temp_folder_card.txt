export interface FolderCardProps {
	folder: any; // Temporarily use any to resolve immediate error
	onClick?: (folderData: any) => void;
	href?: string;
	className?: string;
	interactive?: boolean;
	tcgMode?: boolean;
}

export const FolderCard = memo(function FolderCard({
	folder,
	onClick,
	href,
	className,
	interactive = true,
	tcgMode = false,
}: FolderCardProps) {
	// Workaround for Biome's unused parameter lint rule
	// These parameters are used in JSX or passed to sub-components
	void onClick;
	void href;
	void className;
	void interactive;
	void tcgMode;

	// Preparar datos con fallbacks
	const { data: recentImagesData } = useRecentFolderImages(folder?.id || '');
	const { data: folderStatsData } = useFolderStats(folder?.id || '');

	// Validar que el objeto folder exista
	if (!folder) {
		console.error('FolderCard recibió un objeto folder inválido');
		return null;
	}
});
